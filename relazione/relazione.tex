\section{Introduction}
This report presents a simple study of the quantum harmonic oscillator by means of
Markov Chain Monte Carlo (MCMC) simulations, using the path integral formulation.
We estimate the thermal averages of a few relevant physical observables and the
first energy gaps, making use of suitable interpolating operators and of their
connected correlators. Finally, we reconstruct the squared modulus of the
ground-state wave function and compare it with the analytic solution.

\subsection{The harmonic oscillator}
The Hamiltonian of the harmonic oscillator is
\begin{equation}
    H = -\frac{\hbar^2}{2m} \frac{d^2}{dx^2} + \frac{1}{2}m\omega^2x^2
\end{equation}
whose eigenstates in the position representation are
\begin{equation}
    \psi_n(x) = \frac{1}{\sqrt{2^n n!}} \left( \frac{m\omega}{\pi\hbar} \right)^{1/4} e^{-\frac{m\omega x^2}{2\hbar}} H_n\left( \sqrt{\frac{m\omega}{\hbar}} x \right)
\end{equation}
where the $H_n$ are the Hermite polynomials. The corresponding energy eigenvalues are
\begin{equation}
    E_n = \hbar\omega \left( n + \frac{1}{2} \right)
    \label{eq:energy_levels}
\end{equation}

\subsection{Quantum statistics and the Euclidean path integral}
The partition function of a quantum system is computed as
\begin{equation}
    Z = \text{Tr}(e^{-\beta H})
\end{equation}
To make the connection with the path integral for real-time propagation explicit,
it can be written as
\begin{equation}
    Z = \int dx \bra{x} e^{-\beta H} \ket{x}
\end{equation}
We can apply the same path integral procedure by performing the substitution
$t \to -i\tau = -i\beta$, moving to imaginary time through the Wick rotation
(the only difference lies in the need to impose periodic boundary conditions).
In this formulation the partition function, in the continuum limit, becomes
(with $\hbar=1$)
\begin{equation}
    Z \propto \int_{x(0)=x(\beta)} \mathcal{D}x(\tau) e^{-S_E[x(\tau)]}
\end{equation}
where $S_E$ is the Euclidean action, given by
\begin{equation}
    S_E[x(\tau)] = \int_0^\beta d\tau' \left[ \frac{1}{2} m \left( \frac{dx}{d\tau'} \right)^2 + V(x(\tau')) \right]
\end{equation}
At this point we introduce the thermodynamic average of an operator $O$:
\begin{equation}
    \langle O \rangle = \sum_{n} P_{n} O_{n}
\end{equation}
where
\begin{equation*}
    P_{n} = \frac{e^{-\beta E_n}}{Z} \quad\quad\quad O_{n} = \bra{n} O \ket{n}
\end{equation*}
so that
\begin{equation}
    \langle O \rangle = \frac{\text{Tr}(O e^{-\beta H})}{Z}
\end{equation}
Applying the path integral procedure to the numerator in the same way, we obtain
\begin{equation}
    \langle O \rangle = \frac{\int \mathcal{D}x(\tau) e^{-S_E[x(\tau)]} O[x(\tau)]}{\int \mathcal{D}x(\tau) e^{-S_E[x(\tau)]}}
\end{equation}
Thermal averages are therefore obtained as averages over all periodic paths,
weighted by the exponential of the Euclidean action. We can estimate these values
by designing a Markov chain that samples the space of paths according to the above
distribution. In the numerical simulation the paths are discretised into $N_t$ time
slices, and the continuum limit can be analysed by decreasing the time step
$\frac{\beta}{N_t}$ at fixed $\beta$.

\section{Markov chain}
The discrete Euclidean action for a path $x_i$ with $i=0,1,\dots,N-1$ is
\begin{equation}
    S_E^{(L)}[x_i] = a \sum_{i=0}^{N_t-1} \left[ \frac{m\omega^2 x_i^2}{2} + \frac{m}{2} \left(\frac{x_{i+1}-x_i}{a}\right)^2 \right]
\end{equation}
It is convenient to move to dimensionless units by setting
\begin{equation}
    \eta = a \omega \quad\quad\quad y_i = \frac{x_i}{l} \quad\quad\quad l = \sqrt{\frac{1}{m\omega}}
\end{equation}
where $\frac{1}{\omega}$ is the characteristic time scale of the system. The action
then becomes
\begin{equation}
    S_E^{(L)} = \sum_{i=0}^{N_t-1} \left[ y_i^2 \left( \frac{\eta}{2} + \frac{1}{\eta}\right) - \frac{1}{\eta} y_i y_{i+1}\right]
\end{equation}
The probability distribution to be sampled is therefore
\begin{equation}
    P(y_0,\dots,y_{N_t-1}) = \frac{\exp\left\{ - \sum_{i=0}^{N_t-1} \left[y_i^2 \left(\frac{\eta}{2} + \frac{1}{\eta}\right) - \frac{1}{\eta} y_i y_{i+1}\right]\right\}}{\int \prod_{j=0}^{N_t-1} dy_j \exp\left\{ - \sum_{i=0}^{N_t-1} \left[y_i^2 \left(\frac{\eta}{2} + \frac{1}{\eta}\right) - \frac{1}{\eta} y_i y_{i+1}\right]\right\}}
\end{equation}
We implemented a chain made of 1 heat-bath sweep and 5 over-relaxation
(microcanonical) sweeps which, combined together, guarantee aperiodicity and
irreducibility.

\subsection{Heat-bath update}
The local heat-bath algorithm determines the new value $y_i^{t}$ of the path
independently of the value $y_i$ at the previous step, by sampling directly from
the conditional distribution $P(y_i | y_{j \neq i})$. In our case this algorithm is
particularly efficient, since that distribution is simply a Gaussian centred at
\begin{equation}
    \mu_i = \frac{\gamma}{2\alpha} \quad\quad\quad \gamma = \frac{y_{i-1} + y_{i+1}}{\eta} \quad\quad\quad \alpha = \frac{\eta}{2} + \frac{1}{\eta}
    \label{eq:hb_params}
\end{equation}
which depends only on the values of the adjacent points $y_{i-1}$ and $y_{i+1}$.
The variance is
\begin{equation}
    \sigma^2 = \frac{1}{2\alpha}
\end{equation}
The acceptance for the new state is equal to 1. The Box-Muller algorithm is used to
sample this Gaussian.

\subsection{Microcanonical update}
The microcanonical (over-relaxation) update is a deterministic update that preserves
the Euclidean action, so that the acceptance probability is always 1. The new value
$y_i^{t}$ is computed as
\begin{equation}
    y_i^{t} = 2\mu - y_i
    \label{eq:overrelax}
\end{equation}
where $\mu$ is the conditional mean defined above (equation~\ref{eq:hb_params}).
That is, we are reflecting the point $y_i$ about the mean of the Gaussian. This kind
of update is particularly effective at reducing autocorrelations between samples and
is computationally cheap.

\section{Simulation and data collection}

\subsection{Simulation parameters}
In order to study the behaviour of the energy as a function of temperature and to
verify the agreement with theory, we fixed $\eta = 0.1$ and varied $\beta\hbar\omega$
from 1 to 20. We then fixed $\beta\hbar\omega = 10$, using values of $N_t$ from 4 to
200, in order to study the continuum limit of the observables and the first three
energy gaps, estimated through the connected correlators of $y$, $y^2$, $y^3$ and
$A$ (using only the largest values of $N_t$).

We fixed the number of Markov chain steps to $10^7$, sampling the observables every
10 steps. The particular combination of heat bath, over-relaxation and sampling
frequency resulted in very short autocorrelation times (of order unity, up to a
maximum of a few tens) and a very small $\tau_{exp}$.
We chose a hot start, initialising the path with random values drawn from a uniform
pdf on $[-1,1]$ and discarding, for safety, $5\times 10^5$ thermalisation steps.

\subsection{Measured observables}
We measured $\langle H \rangle$, $\langle y \rangle$, $\langle y^2 \rangle$,
$\langle y^3 \rangle$ and $\langle A \rangle$, where $A$ is the interpolating
operator of the third excited state of the oscillator, defined by
\begin{equation}
    A = y^3 - \frac{3}{2} y
    \label{eq:operator_A}
\end{equation}
We also measured the connected correlators
\begin{equation}
    C_O(n) = \langle O(n) O(0) \rangle - |\langle O \rangle|^2
    \label{eq:connected_corr}
\end{equation}
for $n=0,1,\dots,\frac{N_t}{2}$, for the operators $\langle y \rangle$,
$\langle y^2 \rangle$, $\langle y^3 \rangle$ and $\langle A \rangle$.

\subsection{Estimators used}
The estimates of $\langle y \rangle$, $\langle y^2 \rangle$, $\langle y^3 \rangle$
and $\langle A \rangle$ are obtained as simple averages of the corresponding
observables (which depend only on the position) over the sampled paths,
\begin{equation}
\overline{y} = \frac{1}{N_t} \sum_{i=1}^{N_t} y_i, \quad
\overline{y^2} = \frac{1}{N_t} \sum_{i=1}^{N_t} y_i^2, \quad
\overline{y^3} = \frac{1}{N_t} \sum_{i=1}^{N_t} y_i^3, \quad
\overline{A} = \frac{1}{N_t} \sum_{i=1}^{N_t} \left(y_i^3 - \frac{3}{2} y_i\right)
\end{equation}
The expectation values of $y$, $y^3$ and $A$ all vanish by symmetry.
The energy is computed as
\begin{equation}
    \overline{H} = \left\langle \frac{1}{2 N_t} \sum_{i=0}^{N_t-1} y_i^2 + \frac{1}{2\eta} - \frac{1}{2N_t} \sum_{i=0}^{N_t-1} \frac{\left(y_{i+1} - y_i\right)^2}{\eta^2} \right\rangle
    \label{eq:energy_estimator}
\end{equation}
to be compared with its theoretical value which, in dimensionless units, coincides
with the value of $\langle y^2 \rangle$ and reads
\begin{equation}
    \langle H \rangle = \langle y^2 \rangle =\frac{1}{2} \frac{1 + e^{-\beta\hbar\omega}}{1 - e^{-\beta\hbar\omega}}
    \label{eq:theory_energy}
\end{equation}
The estimates of the correlators are obtained from
\begin{equation}
    \overline{O(n)O(0)} = \frac{1}{N_t} \sum_{i=0}^{N_t-1} O(x_i) O(x_{i+n})
\end{equation}

\section{Analysis and results}

procedure. The uncertainties on the connected correlators and on the energy gaps
were estimated using a blocked jackknife, looking for the plateau in order to obtain
a reliable value while handling the autocorrelation between the data. As an
illustration, figures \ref{fig:blocking_uncertainties} and
\ref{fig:jackknife_uncertainties} show the behaviour of the uncertainties as a

It should be noted that this procedure is not rigorous, since the energy data used
are not the appropriate extrapolations to the continuum limit. Having fixed the step
$\eta$ exposes us to finite-step deviations of order $O(\eta^2)$, and probably to
values of $N_t$ that are too small for the smallest values of $\beta\hbar\omega$.
For a more reliable result one could adopt the approach described in the following
section, using several values of $N_t$ for each $\beta\hbar\omega$ and performing a
linear fit in $\eta^2$:
\begin{equation}
    E(\eta^2) = a + b \eta^2
    \label{eq:linear_model}
\end{equation}
In this way one extrapolates the continuum limit of the energy for each value of
$\beta\hbar\omega$, and the fit as a function of temperature can be performed using
those values, corresponding to the intercept of the linear fit.
Because of limited computational resources and insufficient statistics, we proceeded
as described above anyway, assuming that $\eta=0.1$ is small enough not to introduce
significant deviations from the continuum-limit model.
The fit gives a value of $a$ that differs from the theoretical value of 0.5 by almost
4 sigma, while $b$ is well compatible (table~\ref{tab:risultati_fit_evst}).

\subsection{Energy gaps}
To extract the energy gaps we resort to the effective gap which, in the
dimensionless case, is defined by
\begin{equation}
    \Delta E(n) = \frac{1}{\eta} \log\left[\frac{C(n)}{C(n+1)}\right]
    \label{eq:gap_efficace}
\end{equation}
Indeed, for large values of $n$ the connected correlator behaves as
\begin{equation}
    C(n) \sim e^{-n \eta \Delta E }
    \label{eq:corr_asymptotic}
\end{equation}
The extracted $\Delta E$ refers to the first excited level $E_{\overline{m}}$ such
that
\begin{equation}
    \bra{0} O \ket{\overline{m}} \neq 0
    \label{eq:selection_rule}
\end{equation}
Using the definition above (equation~\ref{eq:gap_efficace}) we obtain a sequence of
estimates of the energy gap as a function of $n$, which should flatten around a
plateau corresponding to the gap.

\begin{itemize}
    \item \textbf{Gap $E_1 - E_0$}: to estimate the first energy gap we can use
    either the connected correlator of $y$ or that of $y^3$, but the former is more
    accurate, since it also satisfies
\begin{equation}
    \langle 0 | y | n \rangle = \frac{1}{\sqrt{2}} \delta_{n,1}
    \label{eq:y_matrix_element}
\end{equation}
whereas $y^3$ has non-vanishing overlap with $0$ and $3$ as well, so that the gap
profile converges slowly to the plateau, and before it is reached in a stable way
the data become very noisy (since the correlator reaches very small values and the
ratio becomes unstable, with enormous uncertainties).
    \item \textbf{Gap $E_2 - E_0$}: to estimate the second energy gap we use the
    connected correlator of $y^2$, which has non-vanishing overlap only with $0$ and
    $2$ (and is therefore particularly well suited as an interpolating operator for
    this gap).
    \item \textbf{Gap $E_3 - E_0$}: to estimate the third energy gap we use the
    connected correlator of $A$, but a priori this is only feasible for the harmonic
    oscillator, since we know its analytic solution. Indeed $A$
    (equation~\ref{eq:operator_A}) is built ad hoc to be proportional to the Hermite
    polynomial $H_3$, which we know to be the eigenfunction of the third excited
    state. It is not possible, on the other hand, to use $y^3$ as an interpolating
    operator for the third gap, since it also has non-vanishing overlap with $1$.

\end{itemize}

state alone, since the temperature is too high to isolate it.


\section{Conclusions}

Given the results obtained, and knowing the analytic solution of the oscillator and
its thermodynamics, we can reasonably conclude that the Markov chain implemented
here samples the probability distribution of the paths correctly, since the
estimates of the primary observables and of the energy gaps are compatible with the
expected theoretical values, and since under the appropriate conditions we are able
to adequately reconstruct the squared modulus of the ground-state wave function.
By increasing the statistics one could obtain more precise results for extracting
the values of the observables in the continuum limit, also allowing more accurate
analyses.
